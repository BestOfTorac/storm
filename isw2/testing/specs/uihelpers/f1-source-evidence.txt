  141:         int uptime = Integer.parseInt(val);
  142:         LinkedList<String> tmp = new LinkedList<>();
  143:         for (Object[] divider : dividers) {
  144:             if (uptime > 0) {

  146:                 Integer div = (Integer) divider[1];
  147:                 if (div != null) {
  148:                     tmp.addFirst(uptime % div + state);

  150:                 } else {
  151:                     tmp.addFirst(uptime + state);

  153:             }
  154:         }
